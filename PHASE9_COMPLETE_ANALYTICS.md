## 💾 DATABASE SCHEMA ADDITIONS

```sql
CREATE TABLE daily_metrics (
  id UUID PRIMARY KEY,
  user_id UUID,
  metric_name TEXT,
  value DECIMAL,
  date DATE,
  created_at DATETIME
);

CREATE TABLE anomalies (
  id UUID PRIMARY KEY,
  user_id UUID,
  metric TEXT,
  value DECIMAL,
  zscore DECIMAL,
  severity TEXT,
  detected_at DATETIME
);

CREATE TABLE dashboards (
  id UUID PRIMARY KEY,
  user_id UUID,
  name TEXT,
  description TEXT,
  widgets JSONB,
  layout TEXT,
  is_public BOOLEAN,
  created_at DATETIME
);

CREATE TABLE insights (
  id UUID PRIMARY KEY,
  user_id UUID,
  insights JSONB,
  alerts JSONB,
  opportunities JSONB,
  recommendations JSONB,
  generated_at DATETIME
);

CREATE TABLE alerts (
  id UUID PRIMARY KEY,
  user_id UUID,
  type TEXT,
  severity TEXT,
  message TEXT,
  action_required BOOLEAN,
  is_read BOOLEAN,
  created_at DATETIME
);

CREATE TABLE scheduled_reports (
  id UUID PRIMARY KEY,
  user_id UUID,
  report_id TEXT,
  schedule TEXT,
  recipients JSONB,
  is_active BOOLEAN,
  next_delivery DATETIME,
  created_at DATETIME
);
```
